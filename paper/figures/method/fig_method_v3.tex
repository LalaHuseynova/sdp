% one batch of four records: #1,#2 = x,y of the centre; #3..#6 = u (audited
% unsafe root), w (withheld), k (kept)
\newcommand{\bt}[6]{%
  \foreach \c [count=\i from 0] in {#3,#4,#5,#6}{%
    \begin{scope}[shift={({#1+(\i-1.5)*0.31},{#2})}]
      \node[rec\c] {};
      \if\c u \draw[white, line width=0.6pt] (-0.7mm,-0.7mm) -- (0.7mm,0.7mm) (-0.7mm,0.7mm) -- (0.7mm,-0.7mm);\fi
      \if\c w \draw[fmUnsafe, line width=0.5pt] (-0.8mm,-0.8mm) -- (0.8mm,0.8mm);\fi
    \end{scope}%
  }%
}
% a context column (rounded box, label, two batch rows) at x = #1, y-top = 0
\newcommand{\cx}[4]{% #1 x, #2 label, #3 row-1 pattern {....}, #4 row-2 pattern {....}
  \node[ctx, minimum width=1.5cm, minimum height=1.3cm] (c#2-#1) at (#1,-0.725) {};
  \node[anchor=south, font=\sffamily\tiny, text=black!70] at (#1,-0.06) {context #2};
}
\newcommand{\panel}[1]{\begin{scope}[xshift=#1]}

% ---- panel positions: x0 = 0, 5.85, 11.7 (cm); contexts at +0.85, +2.6, +4.35 ----

}
